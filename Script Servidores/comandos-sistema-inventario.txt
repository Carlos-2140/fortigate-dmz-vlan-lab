COMANDOS UTILIZADOS - SISTEMA DE INVENTARIO
Ubuntu Server 24.04
IP: 10.21.41.3/28
Gateway: 10.21.41.1

# Verificar interfaz de red
ip a

# Cambiar hostname
sudo hostnamectl set-hostname Sistema-Inventario

# Actualizar repositorios
sudo apt update

# Instalar paquetes necesarios
sudo apt install -y nginx openssh-server curl wget iputils-ping net-tools

# Habilitar y arrancar Nginx
sudo systemctl enable --now nginx

# Habilitar y arrancar SSH
sudo systemctl enable --now ssh

# Verificar estado de Nginx
sudo systemctl status nginx

# Verificar estado de SSH
sudo systemctl status ssh

# Crear página web del Sistema de Inventario
echo '<h1>Sistema de Inventario</h1><p>Laboratorio FortiGate</p>' | sudo tee /var/www/html/index.html

# Editar configuración de red estática
sudo nano /etc/netplan/50-cloud-init.yaml

# Configuración usada en Netplan:
# network:
#   version: 2
#   ethernets:
#     ens3:
#       dhcp4: false
#       addresses:
#         - 10.21.41.3/28
#       routes:
#         - to: default
#           via: 10.21.41.1
#       nameservers:
#         addresses:
#           - 8.8.8.8
#           - 1.1.1.1

# Aplicar configuración de red
sudo netplan apply

# Verificar rutas
ip route

# Probar gateway de la DMZ
ping -c 4 10.21.41.1

# Verificar que Nginx escucha en el puerto 80
sudo ss -lntp | grep :80

# Verificar que SSH escucha en el puerto 22
sudo ss -lntp | grep :22

# Probar servicio web local
curl http://127.0.0.1

# Probar servicio web por su IP
curl http://10.21.41.3

# Revisar repositorios configurados de Ubuntu
grep -R "^URIs:" /etc/apt/sources.list /etc/apt/sources.list.d/ 2>/dev/null
cat /etc/apt/sources.list.d/ubuntu.sources

# Probar actualizaciones permitidas
sudo apt update
