COMANDOS UTILIZADOS - SISTEMA DE CAJA
Ubuntu Server 24.04
IP: 10.21.41.2/28
Gateway: 10.21.41.1

# Verificar interfaz de red
ip a

# Cambiar hostname
sudo hostnamectl set-hostname Sistema-Caja

# Actualizar repositorios
sudo apt update

# Instalar paquetes necesarios
sudo apt install -y nginx openssh-server curl wget iputils-ping net-tools

# Habilitar y arrancar Nginx
sudo systemctl enable --now nginx

# Habilitar y arrancar SSH
sudo systemctl enable --now ssh

# Verificar estado de Nginx
sudo systemctl status nginx

# Verificar estado de SSH
sudo systemctl status ssh

# Crear página web del Sistema de Caja
echo '<h1>Sistema de Caja</h1><p>Laboratorio FortiGate</p>' | sudo tee /var/www/html/index.html

# Editar configuración de red estática
sudo nano /etc/netplan/50-cloud-init.yaml

# Configuración usada en Netplan:
# network:
#   version: 2
#   ethernets:

# Aplicar configuración de red
sudo netplan apply

# Verificar rutas
ip route

# Probar gateway de la DMZ
ping -c 4 10.21.41.1

# Probar servicio web local
curl http://127.0.0.1

curl http://10.21.41.2

# Probar acceso al servidor de Inventario desde la misma DMZ
